{{- $page := . -}} {{- $content := .RawContent -}} {{- $pagesByKey := hugo.Data.wiki_links | default dict -}} {{- range uniq (findRE `\[\[[^\]\n]+\]\]` $content) -}} {{- $link := . -}} {{- $value := replaceRE `^\[\[([^\]\n]+)\]\]$` "$1" $link -}} {{- $parts := split $value "|" -}} {{- $key := strings.TrimSpace (index $parts 0) -}} {{- $targetPage := index $pagesByKey $key -}} {{- $label := $key -}} {{- if gt (len $parts) 1 -}} {{- $label = delimit (after 1 $parts) "|" -}} {{- else if $targetPage -}} {{- $label = $targetPage.title -}} {{- end -}} {{- $escapedLabel := replace $label `[` `\[` -}} {{- $escapedLabel = replace $escapedLabel `]` `\]` -}} {{- if $targetPage -}} {{- $targetURL := strings.TrimPrefix "/" $targetPage.url | relURL -}} {{- $content = replace $content $link (printf "[%s](%s)" $escapedLabel $targetURL) -}} {{- else if $page.Params.wiki_projection -}} {{- $content = replace $content $link $escapedLabel -}} {{- end -}} {{- end -}} {{ .RenderString $content }}
